# Parameter checks run by build2lynx.sh through tools/clap-smoke after bundling.
# Format: <description> | <clap-smoke options>
# The input is a 1 kHz sine (0.5 amplitude, both channels); the sidechain
# input gets silence. "Peak" (--expect-db) shows the gain recovers to unity
# between dips; "trough" (the quietest 10 ms window) shows how far it dips.
# Expected values come from the curve maths (crates/dsp-core/src/duck), not
# from measuring the plugin.

# --- Tempo mode (the default; 1/4, depth 100 %, release 60 %, tension +40 %) ---
ducks on the beat, recovers fully    | --tempo 120 --expect-db 0 --trough-max-db -30
depth 50 %: floor is -6.02 dB        | --tempo 120 --set "Depth=50" --expect-db 0 --expect-trough-db -5.95 --tol 0.15
depth 0: untouched                   | --tempo 120 --set "Depth=0" --expect-db 0 --expect-trough-db 0
mix 0: untouched                     | --tempo 120 --set "Mix=0" --expect-db 0 --expect-trough-db 0
1/8 notes still duck                 | --tempo 120 --set "Rate=1/8" --expect-db 0 --trough-max-db -30
long release carries over (90 BPM)   | --tempo 90 --set "Hold=50" --set "Release=100" --expect-db -12.76 --tol 0.1 --trough-max-db -30
no transport: runs at 120 BPM        | --expect-db 0 --trough-max-db -30
stopped, run-when-stopped off: unity | --set "Run When Stopped=Off" --expect-db 0 --expect-trough-db 0

# --- MIDI mode (Hold 0 ms, Release 300 ms; notes on key 36, input tone held) ---
midi: dips on each note, recovers      | --set "Mode=MIDI" --note 36 --hold --note-every 24000 --expect-db 0 --trough-max-db -30
midi: fast notes retrigger mid-release | --set "Mode=MIDI" --note 36 --hold --note-every 12000 --expect-db -3.99 --tol 0.15 --trough-max-db -30
midi: velocity 100 %, soft notes       | --set "Mode=MIDI" --set "Velocity=100" --note 36 --velocity 0.5 --hold --note-every 24000 --expect-db 0 --expect-trough-db -5.95 --tol 0.15
midi: velocity 0 %, soft notes = full  | --set "Mode=MIDI" --note 36 --velocity 0.25 --hold --note-every 24000 --expect-db 0 --trough-max-db -30
midi: no notes, untouched              | --set "Mode=MIDI" --tempo 120 --expect-db 0 --expect-trough-db 0
midi: hold 100 ms widens the dip       | --set "Mode=MIDI" --set "Hold (ms)=100" --note 36 --hold --note-every 24000 --expect-db 0 --trough-max-db -40

# --- Audio mode (sidechain: silence, or --sidechain-kicks: a 60 Hz kick on each beat;
# detector defaults: threshold -24 dB, 80 Hz, retrigger 60 ms; Release 300 ms). Listen's level
# (+1.46 dB over the input) comes from simulating the band-pass on the kick. ---
audio: dips on each kick, recovers       | --set "Mode=Audio" --sidechain-kicks 120 --expect-db 0 --trough-max-db -30
audio: silent sidechain, untouched       | --set "Mode=Audio" --tempo 120 --expect-db 0 --expect-trough-db 0
audio: threshold above the kicks         | --set "Mode=Audio" --set "SC Threshold=0" --sidechain-kicks 120 --expect-db 0 --expect-trough-db 0
audio: band-pass at 400 Hz misses them   | --set "Mode=Audio" --set "SC Freq=400" --set "SC Threshold=-12" --sidechain-kicks 120 --expect-db 0 --expect-trough-db 0
audio: listen plays the band-passed kick | --set "Mode=Audio" --set "SC Listen=On" --sidechain-kicks 120 --expect-db 1.46 --tol 0.05 --trough-max-db -40
tempo mode ignores the sidechain         | --set "Depth=0" --sidechain-kicks 120 --expect-db 0 --expect-trough-db 0

# --- Output ---
output -6 dB (on top of the ducking) | --tempo 120 --set "Output=-6" --expect-db -6 --trough-max-db -30
output at its -12 dB limit           | --set "Depth=0" --set "Output=-12" --expect-db -12
